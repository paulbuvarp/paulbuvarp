{{/* Persistent language toggle. Always shows both languages so the reader can see the choice exists, rather than discovering it. */}} {{- $cur := . -}}
{{- range site.Languages -}} {{- $lang := . -}} {{- $target := "" -}} {{- if eq $lang.Lang site.Language.Lang -}} {{- $target = $cur.RelPermalink -}} {{- else -}} {{- range $cur.Translations -}} {{- if eq .Language.Lang $lang.Lang }}{{ $target = .RelPermalink }}{{ end -}} {{- end -}} {{- if not $target }}{{ $target = printf "/%s/" $lang.Lang | replaceRE "^/en/$" "/" }}{{ end -}} {{- end -}} {{ if eq $lang.Lang "en" }}EN{{ else }}NO{{ end }} {{- end -}}
